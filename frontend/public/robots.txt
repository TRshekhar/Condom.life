const fs = require('fs');
const path = require('path');

// This script generates sitemap.xml after build
// Run with: node scripts/generate-sitemap.js

const SITE_URL = 'https://condom.life';

// Static pages
const staticPages = [
  { url: '/', changefreq: 'daily', priority: '1.0' },
  { url: '/rankings', changefreq: 'weekly', priority: '0.9' },
  { url: '/blog', changefreq: 'weekly', priority: '0.7' },
  { url: '/about', changefreq: 'monthly', priority: '0.3' },
  { url: '/contact', changefreq: 'yearly', priority: '0.2' },
];

// Function to generate sitemap XML
function generateSitemap(pages) {
  const lastmod = new Date().toISOString().split('T')[0];
  
  const urls = pages.map(page => `
  <url>
    <loc>${SITE_URL}${page.url}</loc>
    <lastmod>${lastmod}</lastmod>
    <changefreq>${page.changefreq}</changefreq>
    <priority>${page.priority}</priority>
  </url>`).join('');

  return `<?xml version="1.0" encoding="UTF-8"?>
<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">
${urls}
</urlset>`;
}

// Generate and save sitemap
const sitemap = generateSitemap(staticPages);
const buildDir = path.join(__dirname, '..', 'build');
fs.writeFileSync(path.join(buildDir, 'sitemap.xml'), sitemap);

console.log('Sitemap generated successfully!');

// FILE 3: scripts/generate-dynamic-sitemap.js
// ====================================

// This script generates sitemap including dynamic product pages
// You'll need to fetch products from your API

const fs = require('fs');
const path = require('path');
const fetch = require('node-fetch');

const SITE_URL = 'https://condom.life';
const API_URL = process.env.REACT_APP_API_URL || 'http://localhost/condom-reviews/api';

async function generateDynamicSitemap() {
  try {
    // Fetch all products
    const productsResponse = await fetch(`${API_URL}/products.php?limit=1000`);
    const products = await productsResponse.json();

    // Static pages
    const staticPages = [
      { url: '/', changefreq: 'daily', priority: '1.0' },
      { url: '/rankings', changefreq: 'weekly', priority: '0.9' },
      { url: '/blog', changefreq: 'weekly', priority: '0.7' },
    ];

    // Add product pages
    const productPages = products.map(product => ({
      url: `/product/${product.id}/${product.name.toLowerCase().replace(/\s+/g, '-')}`,
      changefreq: 'weekly',
      priority: '0.8'
    }));

    const allPages = [...staticPages, ...productPages];
    const lastmod = new Date().toISOString().split('T')[0];

    const urls = allPages.map(page => `
  <url>
    <loc>${SITE_URL}${page.url}</loc>
    <lastmod>${lastmod}</lastmod>
    <changefreq>${page.changefreq}</changefreq>
    <priority>${page.priority}</priority>
  </url>`).join('');

    const sitemap = `<?xml version="1.0" encoding="UTF-8"?>
<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">
${urls}
</urlset>`;

    const buildDir = path.join(__dirname, '..', 'build');
    fs.writeFileSync(path.join(buildDir, 'sitemap.xml'), sitemap);

    console.log(`Sitemap generated with ${allPages.length} URLs!`);
  } catch (error) {
    console.error('Error generating sitemap:', error);
  }
}

generateDynamicSitemap();
